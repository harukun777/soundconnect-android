cmake_minimum_required(VERSION 3.31)
project(soundconnect_android LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(UPSTREAM "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../upstream/SonyHeadphonesClient")
if(NOT EXISTS "${UPSTREAM}/libmdr/CMakeLists.txt")
    message(FATAL_ERROR "Run git submodule update --init --recursive before building")
endif()
# Build only the portable protocol library, never the desktop Bluetooth backend.
include(FetchContent)
FetchContent_Declare(fmt GIT_REPOSITORY https://github.com/fmtlib/fmt.git GIT_TAG 12.1.0 GIT_SHALLOW TRUE)
set(FMT_INSTALL OFF CACHE BOOL "" FORCE)
set(MDR_BUILD_SHARED OFF CACHE BOOL "" FORCE)
add_compile_definitions(MDR_PLATFORM_ANDROID MDR_ENABLE_LOG=0 FMT_USE_LOCALE=0)
add_subdirectory("${UPSTREAM}/libmdr" "${CMAKE_CURRENT_BINARY_DIR}/mdr")
add_library(soundconnect SHARED bridge.cpp)
target_link_libraries(soundconnect PRIVATE mdr)
target_link_options(soundconnect PRIVATE "-Wl,-z,max-page-size=16384")
